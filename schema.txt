    user_id PK,
	username,	
	email,
	password,
	rating,
	created_at

)




BLOG_POST(
    post_id PK,
	user_id FK → USER(user_id)

    content 
)




BLOG_COMMENT(

    user_id FK → USER(user_id),
    post_id FK → BLOG_POST(post_id),
	text
	
)




TAGS(
	tag_id PK,
	tag_name
)




BLOG_TAG(
    post_id FK → BLOG_POST(post_id),
   	tag_id FK → TAG(tag_id),
   	PK(post_id, tag_id)
)




BLOG_LIKE(

)




BLOG_BOOKMARKS(
    user_id FK → USER(user_id),

)

PROBLEM_BOOKMARKS(
